	\begin{frame}{Introduction to LaTeX}
		
		\begin{itemize}
			\item LaTeX is a document preparation system based on TeX.
			\item It is widely used in academic and technical fields.
			\item It helps create professional and well-structured documents.
			\item It is particularly useful for mathematical and scientific writing.
		\end{itemize}
		
	\end{frame}
	
	\begin{frame}{Features and Applications of LaTeX}
		
		\textbf{Important Features}
		
		\begin{itemize}
			\item Professional document formatting
			\item Mathematical equations and symbols
			\item Tables, figures and references
			\item Automatic numbering and table of contents
		\end{itemize}
		
		\vspace{0.3cm}
		
		\textbf{Applications}
		
		\begin{itemize}
			\item Research papers and academic publications
			\item Technical reports and project documentation
			\item Books, presentations and scientific documents
		\end{itemize}
		
	\end{frame}

		\begin{itemize}

			\item It provides accurate and consistent formatting.
			\item It simplifies the preparation of complex mathematical documents.
			\item It is an important tool for academic and technical communication.
		\end{itemize}
