	\begin{frame}{What is LaTeX}
		
		LaTeX is a document preparation system used for creating
		professional and well-structured documents.
		
		\vspace{0.5cm}
		
		\textbf{Major features:}
		
		\begin{itemize}
			\item Professional typesetting
			\item Mathematical equation support
			\item Automatic numbering
			\item Easy document organization
			\item Customizable formatting
		\end{itemize}
		
	\end{frame}
	
	% Content slide
	\begin{frame}{Document Structure}
		
		A LaTeX document can be organized using:
		
		\begin{itemize}
			\item Chapters
			\item Sections
			\item Subsections
			\item Abstract
			\item Headers and Footers
		\end{itemize}
		
		\vspace{0.5cm}
		
		Packages such as \texttt{geometry}, \texttt{sectsty},
		and \texttt{fancyhdr} provide additional formatting features.
		
	\end{frame}
